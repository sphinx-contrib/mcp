generate_sudoku_puzzle

Generate a new, uniquely-solvable Sudoku puzzle at the given difficulty.

Input schema:

{
  "type": "object",
  "additionalProperties": false,
  "properties": {
    "difficulty": {
      "default": "medium",
      "enum": [
        "very easy",
        "easy",
        "medium",
        "hard",
        "very hard"
      ],
      "type": "string"
    }
  }
}

Output schema:

{
  "description": "Result of generating a new puzzle.",
  "properties": {
    "puzzle": {
      "description": "A 9x9 Sudoku grid; 0 marks an empty cell.",
      "properties": {
        "rows": {
          "description": "9 rows of 9 cells each; 1-9 for a digit, 0 for empty",
          "items": {
            "items": {
              "type": "integer"
            },
            "type": "array"
          },
          "type": "array"
        }
      },
      "required": [
        "rows"
      ],
      "type": "object"
    },
    "difficulty": {
      "enum": [
        "very easy",
        "easy",
        "medium",
        "hard",
        "very hard"
      ],
      "type": "string"
    },
    "givens": {
      "description": "Actual number of filled cells in the generated puzzle",
      "type": "integer"
    }
  },
  "required": [
    "puzzle",
    "difficulty",
    "givens"
  ],
  "type": "object"
}

validate_partial_sudoku_solution

Check whether a partially-filled Sudoku grid is conflict-free and still completable.

Input schema:

{
  "type": "object",
  "additionalProperties": false,
  "properties": {
    "grid": {
      "description": "A 9x9 Sudoku grid; 0 marks an empty cell.",
      "properties": {
        "rows": {
          "description": "9 rows of 9 cells each; 1-9 for a digit, 0 for empty",
          "items": {
            "items": {
              "type": "integer"
            },
            "type": "array"
          },
          "type": "array"
        }
      },
      "required": [
        "rows"
      ],
      "type": "object"
    }
  },
  "required": [
    "grid"
  ]
}

{

    "conflicts": {

{
  "type": "object",
  "additionalProperties": false,
  "properties": {
    "grid": {
      "description": "A 9x9 Sudoku grid; 0 marks an empty cell.",
      "properties": {
        "rows": {
          "description": "9 rows of 9 cells each; 1-9 for a digit, 0 for empty",
          "items": {
            "items": {
              "type": "integer"
            },
            "type": "array"
          },
          "type": "array"
        }
      },
      "required": [
        "rows"
      ],
      "type": "object"
    }
  },
  "required": [
    "grid"
  ]
}

Output schema:

{
  "description": "Result of solving an unsolved grid.",
  "properties": {
    "status": {
      "enum": [
        "satisfiable",
        "conflicting_givens",

      "type": "string"

    "solution": {
      "anyOf": [
        {
          "description": "A 9x9 Sudoku grid; 0 marks an empty cell.",
          "properties": {
            "rows": {
              "description": "9 rows of 9 cells each; 1-9 for a digit, 0 for empty",
              "items": {
                "items": {
                  "type": "integer"
                },
                "type": "array"
              },
              "type": "array"
            }
          },
          "required": [
            "rows"
          ],
          "type": "object"
        },
        {
          "type": "null"
        }
      ]
    },
    "conflicts": {
      "description": "Populated only when status is conflicting_givens",
      "items": {
        "description": "A single cell's position.\n\nrow and col are 1-indexed, matching how Sudoku cells are conventionally described in text,\nrather than the 0-indexed positions used internally.",
        "properties": {
          "row": {
            "description": "1-indexed row",
            "maximum": 9,
            "minimum": 1,
            "type": "integer"
          },
          "col": {
            "description": "1-indexed column",

            "minimum": 1,
            "type": "integer"
          }
        },
        "required": [
          "row",
          "col"
        ],
        "type": "object"
      },
      "type": "array"

  },
  "required": [
